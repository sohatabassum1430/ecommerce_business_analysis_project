TOTAL REVENUE:
DEFINE
    MEASURE 'CLEAN_ORDERS'[Total Revenue] =
        SUMX(
            'CLEAN_ORDERS',
            'CLEAN_ORDERS'[total_amount]
            - 'CLEAN_ORDERS'[discount]
        )

EVALUATE
    SUMMARIZE(
        'CLEAN_ORDERS',
        "Total Revenue", [Total Revenue]
    )

AVG ORDER VALUE:
DEFINE
    MEASURE 'CLEAN_ORDERS'[Total Revenue] =
        SUMX(
            'CLEAN_ORDERS',
            'CLEAN_ORDERS'[total_amount]
            - 'CLEAN_ORDERS'[discount]
        )
    MEASURE 'CLEAN_ORDERS'[Total Orders] =
        COUNTROWS( 'CLEAN_ORDERS' )
    MEASURE 'CLEAN_ORDERS'[Avg Order Value] =
        DIVIDE( [Total Revenue], [Total Orders], 0 )

EVALUATE
    SUMMARIZE(
        'CLEAN_ORDERS',
        "Avg Order Value", [Avg Order Value]
    )


GROSS PROFIT:

    )


Profit margin:

    MEASURE 'CLEAN_ORDER_ITEMS'[Gross Profit] =
        SUMX(
            'CLEAN_ORDER_ITEMS',
            RELATED( 'CLEAN_PRODUCTS'[price] )
            - RELATED( 'CLEAN_PRODUCTS'[cost_price] )
        )
    MEASURE 'CLEAN_ORDERS'[Total Revenue] =
        SUMX(
            'CLEAN_ORDERS',
            'CLEAN_ORDERS'[total_amount]
            - 'CLEAN_ORDERS'[discount]
        )
    MEASURE 'CLEAN_ORDERS'[Profit Margin %] =
        DIVIDE( [Gross Profit], [Total Revenue], 0 ) * 100

    SUMMARIZE(
        'CLEAN_ORDERS',

MOM Revenue growth:
DEFINE
    MEASURE 'CLEAN_ORDERS'[Total Revenue] =
        SUMX(
            'CLEAN_ORDERS',
            'CLEAN_ORDERS'[total_amount]
            - 'CLEAN_ORDERS'[discount]
        )
    MEASURE 'CLEAN_ORDERS'[Revenue Prev Month] =
        CALCULATE(
            [Total Revenue],
            PREVIOUSMONTH( 'CLEAN_ORDERS'[order_date] )
        )
    MEASURE 'CLEAN_ORDERS'[MoM Revenue Growth %] =
        DIVIDE(
            [Total Revenue] - [Revenue Prev Month],
            [Revenue Prev Month], 0
        ) * 100

EVALUATE
    SUMMARIZE(
        'CLEAN_ORDERS',
        "MoM Revenue Growth %", [MoM Revenue Growth %]
    )



Delivery Rate%:	
DEFINE
    MEASURE 'CLEAN_ORDERS'[Total Orders] =
        COUNTROWS( 'CLEAN_ORDERS' )
    MEASURE 'CLEAN_ORDERS'[Delivered Orders] =
        CALCULATE(
            COUNTROWS( 'CLEAN_ORDERS' ),
            'CLEAN_ORDERS'[status] = "Delivered"
        )
    MEASURE 'CLEAN_ORDERS'[Delivery Rate %] =
        DIVIDE( [Delivered Orders], [Total Orders], 0 ) * 100

EVALUATE
    SUMMARIZE(
        'CLEAN_ORDERS',
        "Delivery Rate %", [Delivery Rate %]
